<script>

function generatePlan() {

let name = document.getElementById("name").value;
let age = document.getElementById("age").value;
let weight = document.getElementById("weight").value;
let goal = document.getElementById("goal").value;
let intensity = document.getElementById("intensity").value;
let experience = document.getElementById("experience").value;

if (name === "" || age === "" || weight === "") {
alert("Please fill all the details.");
return;
}

let workouts = [
["Day 1", "Full Body Workout",
"Squats, wall push-ups, lunges and plank."],

["Day 2", "Cardio",
"Walking, light jogging and stretching."],

["Day 3", "Upper Body",
"Wall push-ups, arm raises and shoulder exercises."],

["Day 4", "Active Recovery",
"Light walking and gentle stretching."],

["Day 5", "Lower Body",
"Squats, lunges and glute bridges."],

["Day 6", "Core Workout",
"Plank, bird-dog and gentle core exercises."],

["Day 7", "Rest Day",
"Rest and gentle stretching."]
];

let output = `
<div class="user-info">

<h2>Hello ${name}! 👋</h2>

<p><b>Age:</b> ${age}</p>
<p><b>Weight:</b> ${weight} kg</p>
<p><b>Goal:</b> ${goal}</p>
<p><b>Intensity:</b> ${intensity}</p>
<p><b>Experience:</b> ${experience}</p>

</div>

<h2>📅 Your 7-Day Fitness Plan</h2>
`;

workouts.forEach(function(workout) {

output += `
<div class="day">

<h3>${workout[0]} - ${workout[1]}</h3>

<p>
${workout[2]}
</p>

</div>
`;

});

document.getElementById("result").innerHTML = output;

}

</script>
